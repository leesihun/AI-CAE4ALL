%   ============================================================
%
%   SAOI warpage, cHI-MGNflow SWEEP 3 (KL ladder), stage 1 -- the compressor.
%
%   WHY THIS SWEEP. Sweep 1 came out 0.05-0.47 in sd ratio on every prior arm,
%   with the four prior arms indistinguishable -- so the prior is not the
%   binding constraint. The decoder already reads the geometry through the
%   y-blind skip path, yet at ae_kl_weight 1e-6 nothing stops the encoder from
%   ALSO writing the geometry-determined mean field into z. The latent's
%   variance is then dominated by between-part differences and the 125
%   realizations of one part differ only along tiny directions, which the flow
%   prior's MSE never resolves: it regresses them to the mean and the spread
%   collapses. KL makes every bit in z cost something, and the cheapest bits to
%   drop are the ones the decoder can get for free from the skips -- the mean
%   field. Too much KL drops the realization residual too, so this is a ladder.
%
%   Arm c8k1e4 on board half top: latent_ch 8, ae_kl_weight 1e-4.
%   sweep 1's c8kl, now carried all the way through a prior.
%
%   gpu_ids 0 is LOGICAL: run_sweep3.sh gives each arm its own card through
%   CUDA_VISIBLE_DEVICES.
%
%   ============================================================
model	chi-mgnflow
mode	train_ae  # stage 1 alone; the prior is built but never trained or loaded from here
gpu_ids	0
parallel_mode	ddp
log_file_dir	../../output/chi-mgnflow/saoi_sweep3/ae/top_c8k1e4/ae.log  # own dir per arm: dumps land in <log_dir>/test/0/<epoch>/
modelpath	../../output/chi-mgnflow/saoi_sweep3/top.ae_c8k1e4.pth

%   Dataset
dataset_dir	../../dataset/SAOI/saoi_train_top.h5
split_seed	42
augment_geometry	True

%   Channel layout. The SAOI columns are x y z dx dy dz thickness "Part No.",
%   so rows 0:3 are the reference coordinates (never predicted), 3:6 the
%   displacement state and 6:9 the input-only conditions the model reads every
%   step but never predicts.
input_var	3  # x/y/z displacement -- the state block
output_var	3  # what the model generates
cond_var	3  # input-only rows: thickness and part id
edge_var	8  # deformed dx/dy/dz/dist + reference dx/dy/dz/dist
positional_features	4  # rotation-invariant node features

use_node_types	False
use_world_edges	False
feature_loss_weights	0.1, 0.1, 1  # z_disp is the quantity of interest

%   Hierarchy -- sweep 1's, byte for byte, in every arm. All sixteen training
%   runs share ONE cache directory; the first to arrive builds it under an
%   exclusive lock and the rest block. run_sweep3.sh deletes it at the end.
use_multiscale	True
coarsening_type	voronoi_seedmean
voronoi_clusters	1000, 100
multiscale_levels	2
mp_per_level	4, 6, 8, 6, 4  # V-cycle depth: pre / coarsest / post
hierarchy_variants	4  # rotated across epochs so the model is invariant to the partition
hierarchy_seed	1234
hierarchy_cache_keep	True  # arms finish at different times; the runner removes the dir, not the trainer

%   Isolation from any other sweep on the same box.
%   write_preprocessing False: never opens the shared training HDF5 for
%   writing (normalization travels in the checkpoint, not the file).
%   hierarchy_cache_dir: out of reach of the same-stem prune next to the dataset.
write_preprocessing	False
hierarchy_cache_dir	../../output/chi-mgnflow/saoi_sweep3/mscache/v1000_100
hierarchy_cache_build_workers	8

%   THE SWEPT AXIS. KL is per element (mean over coarse nodes x channels), so
%   its useful range does not move with latent_ch or mesh size.
latent_dim	128  # trunk width, identical across every arm
latent_ch	8
ae_kl_weight	1e-4

training_epochs	2000  # same as sweep 1, so the arms read against it directly

%   Optimization
batch_size	16  # measured b16 < b32 by 0.288 on this data

grad_accum_steps	1

%   Memory / performance

%   Validation. Stage 1 selects on PURE reconstruction (validate_ae_epoch's
