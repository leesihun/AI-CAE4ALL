%   ============================================================

%   TRAINING part 1: the largest condition group of saoi_train_bot.h5,

%
%   The model and its sampling are those of
%   ../config_infer_tok_kl_bot_s26fe_main.txt; only the part differs. The
%   prior trained on this part's realizations, so its draws must reproduce

%   prior's own failure; a match here beside a miss on the unseen s26fe / sm
%   parts is a failure to carry the scatter to a new part.
%
%   infer_dataset is one realization of the part: its geometry and condition
%   rows (the static rollout zeroes the state rows). eval_dataset is every
%   realization of the part, that one included.
%

model	chi-mgnflow
mode	inference
gpu_ids	0  # logical: run_train_parts.sh picks the card
log_file_dir	../../output/chi-mgnflow/saoi_b300_sweep/tok_kl/bot/infer_train_part1.log
modelpath	../../output/chi-mgnflow/saoi_b300_sweep/tok_kl/bot/model.pth
inference_output_dir	../../output/chi-mgnflow/saoi_b300_sweep/tok_kl/bot/infer/train_part1

%   Datasets
infer_dataset	../../dataset/SAOI/train_part1_infer_bot.h5
eval_dataset	../../dataset/SAOI/train_part1_compare_bot.h5  # ground truth for the histogram -- every realization of the part
infer_timesteps	1

%   Channel layout. The SAOI columns are x y z dx dy dz thickness "Part No.",
%   so rows 0:3 are the reference coordinates (never predicted), 3:6 the
%   displacement state and 6:9 the input-only conditions the model reads every
%   step but never predicts.
input_var	3  # x/y/z displacement -- the state block
output_var	3  # what the model generates
cond_var	3  # input-only rows: thickness and part id
edge_var	8  # deformed dx/dy/dz/dist + reference dx/dy/dz/dist
positional_features	4  # rotation-invariant node features

use_node_types	False
use_world_edges	False

%   Hierarchy. The coarsest level (100 clusters) is where the latent lives, so
%   cHI-MGNflow requires use_multiscale True -- a flat stack has no coarse
%   level to compress onto, and that now raises inside
%   LatentDiffusionGraphNet.__init__ rather than training badly.
use_multiscale	True
coarsening_type	voronoi_seedmean
voronoi_clusters	1000, 100
multiscale_levels	2
mp_per_level	4, 6, 8, 6, 4  # V-cycle depth: pre / coarsest / post

%   The hierarchy MUST stay fixed for a whole ODE integration and across every
%   draw of one scene: rebuilding it mid-solve would make each step a
%   different vector field. Four partitions are built up front and rotated
%   across draw batches, so one unlucky partition cannot poison a scene.
hierarchy_seed	0, 1, 2, 3

%   Architecture, repeated from config_train_tok_kl_bot.txt. The checkpoint's own
%   model_config is the authority and overrides these at load time; they are
%   written out so this config describes the run on its own, and so a mismatch
%   prints loudly instead of passing unseen.
latent_dim	128
latent_ch	4
prior_blocks	4
prior_global	token  # one zero-init global node per graph in the prior, exchanging with every coarse node each block
flow_time_freqs	16

%   Sampling. One draw = flow_steps x 2 coarse forwards plus one full-mesh
%   encode/decode, against the part's tens of realizations; 500 draws are
%   ample against so few.
num_vae_samples	500  # draws per scene
flow_steps	30
flow_solver	heun

%   The histogram: generated vs ground-truth z_disp peak-to-valley, one value
%   per draw against one per eval sample. rollout.py writes
%   histogram_compare.png and spread_values.npz under inference_output_dir and
%   prints the [SPREAD] summary lines to the log. This is the product of the
%   run -- there is no separate scoring stage.
make_histogram	True
show_histogram	False  # headless box: save the PNG, never open a viewer
histogram_bins	60
save_rollouts	False  # the spreads still reach spread_values.npz; per-draw HDF5s would be tens of thousands of files
