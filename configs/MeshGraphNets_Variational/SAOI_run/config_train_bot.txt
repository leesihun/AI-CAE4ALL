%   This is the surviving cell of the retired eight-arm FM-v2 ablation
%   (bot/top x P0-P3). P0 -- every flow-matching key at its native default --
%   won sd_ratio, PITtails and PIT_KS in BOTH halves, so there is no FM-v2
%   override tail here: widening the velocity MLP 256 -> 512 was the largest
%   replicated effect in the sweep and it was HARMFUL, and residual FiLM was
%   a clean null. The schedule below (1000 epochs, freeze at 700,
%   prior_grad_to_encoder 0, best_by crps) is the one the ablation was
%   measured under; changing it makes this run incomparable to the evidence
%   that picked the recipe.
%
%   Architecture is the b8_all winner cell scaled DOWN for the larger SAOI
%   meshes: Latent_dim 128, voronoi 1000/100, mp_per_level 4,6,8,6,4.

%   Common params
%   Dataset columns: x y z dx dy dz stress thickness Part-No -- stress,
%   thickness and Part No. are known per-node inputs the model never predicts,
%   so they live in cond_var (input-only), not input_var/output_var (state).
input_var	3  # number of input variables: x_disp, y_disp, z_disp (state: input AND output)

positional_features	4  # Rotation-invariant node features: [centroid_dist, mean_edge_len, + encoding]

%   Network parameters
message_passing_num	15  # Only used for non-multiscale modes (vanilla MeshGraphNets only)
Training_epochs	1000  # Complete from-scratch schedule

num_workers	4  # number of workers for dataloader
std_noise	0.01  # Gaussian physical-input noise (training only)

grad_accum_steps	1  # Gradient accumulation steps

%   Memory / performance
use_checkpointing	True  #faster; set True if OOM

%   Multiscale (Multiscale HI-MGN architecture)
use_multiscale	True  # Enable HI-MGN architecture
coarsening_type	voronoi_seedmean  # Use Voronoi_seedmean only for production, legacy codeset for comparison
voronoi_clusters	1000, 100  # Cluster count per multiscale level (Voronoi coarsening)
multiscale_levels	2  # Number of coarsening levels
mp_per_level	4, 6, 8, 6, 4  # Message-passing blocks per level: down arm, coarsest, up arm (2*levels+1 values)
%   Coarsening-invariance: FPS is seeded per sample id, so training would
%   otherwise see ONE fixed Voronoi partition for every epoch while inference
%   builds a fresh one it has never seen. K variants are built once at cache
%   construction (NOT per epoch) and rotated across epochs; validation stays
%   pinned to variant 0 so its loss stays comparable.
hierarchy_variants	4  # Independently-seeded coarsening partitions cached per sample
hierarchy_cache_keep	True  # Keep the cache; the infer runs rebuild the same hierarchies

%   VAE (MMD-InfoVAE)
use_vae	True  # Enable posterior encoder + stochastic latent path (variational losses)
vae_latent_dim	16  # Posterior/global latent width (must match checkpoint)
vae_mp_layers	5  # Posterior graph-encoder message-passing depth
recon_loss	mse  # huber or mse reconstruction objective
alpha_recon	1000  # Reconstruction-loss multiplier
lambda_mmd	1  # Aggregate-posterior MMD loss multiplier
mmd_bandwidth	median  # MMD bandwidth mode (fixed or median)
vae_graph_aware	True  # Include graph/input context when encoding the posterior
beta_aux	10  # Auxiliary latent-statistics prediction loss weight
posterior_min_std	0.05  # Posterior standard-deviation floor
z_conditioning	adaln  # AdaLN-Zero z modulation (identity at init) instead of the
%   legacy concat fuser Linear([x, z]). The fuser could only add ONE constant
%   vector per graph and, being a bare non-residual Linear between all 28
%   processor blocks, compounded a ~1.33x gain per block -- inflating the
%   residual stream ~10^3 and burying the V-cycle unit-scale unpool branch
%   under the inflated skip branch. Renames parameters, so it is persisted in
%   the checkpoint model_config and inference reads it from there.
mmd_gather_ranks	True  # All-gather z over DDP ranks before MMD (no-op on one card)

%   Prior - gnn_e2e joint training; prior_family fm = conditional flow matching.
%   Every FM-v2 key below is at its native default on purpose: that is the P0
%   cell, and it beat all three proposed upgrades.
prior_type	gnn_e2e  # gnn_e2e enables the jointly trained graph-conditional prior
use_conditional_prior	True  # Use learned conditional prior instead of N(0,I)
prior_family	fm  # fm (flow matching) or gmm
prior_nll_weight	1.0  # Prior density/flow-matching objective multiplier
prior_grad_to_encoder	0.0  # FM may fit q(z|y,c), but may not shrink the encoder
%   target. Opening this lets the flow pull the posterior toward itself, a
%   trivial-solution risk; the ablation ran closed, so the verdict holds at 0.0.
prior_freeze_epoch	700  # Freeze encoder/decoder at 700; the last 300 epochs fit
%   ONLY the prior, on a fresh cosine schedule.
prior_fm_steps	30  # Integration steps for FM-prior sampling/validation
prior_fm_solver	heun  # 2nd-order trapezoid predictor-corrector. Euler O(dt^2)
%   local error overshoots into the z tails exactly where the velocity is most
%   curved (t -> 1) -- the Gen_max spike that had forced steps 20 -> 100.
%   Heun at 30 steps is more accurate than Euler at 100 and ~40% cheaper.
prior_mp_layers	5  # Prior message-passing depth
prior_hidden_dim	256  # Conditional-prior hidden width
prior_temperature	1.0  # Sampling spread control (inference only)

%   Time integration (docs/CONFIGURATION.md sections 3.5 and 6.7)
time_integration	ar_ot  # ar_ot = one-step/teacher-forced (default); ar_rt = full-trajectory autoregressive rollout training
