%   THE INFLATION CURVE -- the free axis.
%   A LIST is cycled batch by batch, so ONE inference pass yields the whole
%   lam-vs-spread curve; every generated spread is tagged with the lam that
%   produced it in spread_values.npz (field `gen_lam`), and rollout.py prints a
%   per-lam `[SPREAD] GEN lam=...` line. lam=1.0 is the honest sample and is
%   what the arms are ranked on; the rest measure how far post-hoc rescaling
%   can be pushed. ConditionalFMPrior.sample_n applies it AFTER integration as
%   z <- c + lam*(z - c) around the per-graph center c, which is the clean
%   width knob (prior_temperature is not: it scales the START of the ODE, which
%   leaves the trajectories the velocity field was trained on).
%   Prior art from the retired campaign: 2.5x on z bought 1.35x on the field.
%   Sub-linear, so this curve is expected to fall short of 1.0 -- it is here to
%   quantify the shortfall per arm, not to fix it.
latent_inflation	1.0, 1.25, 1.5, 2.0, 2.5, 3.0
prior_temperature	1.0  # left at 1.0 on purpose; see above

%   Histogram. NOTE: the PNG pools every lam into one generated histogram and
%   is therefore NOT the deliverable for a multi-lam run -- spread_values.npz
%   is. analyze_sweep.py splits on gen_lam and never reads the PNG.

save_rollouts	False  # 3000 field dumps per arm per eval set is 27x the disk
%   budget of the whole campaign; the npz carries every number that is scored.
